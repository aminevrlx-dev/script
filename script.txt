local function SetSpeed(value)

    value = tonumber(value)

    if not value then
        value = Speed
    end

    value = math.clamp(math.floor(value + 0.5), MIN_SPEED, MAX_SPEED)

    Speed = value

    SpeedBox.Text = tostring(value)

    local Percent = (value - MIN_SPEED) / (MAX_SPEED - MIN_SPEED)

    SliderFill.Size = UDim2.new(Percent, 0, 1, 0)
    Knob.Position = UDim2.new(Percent, 0, 0.5, 0)
end

SpeedBox.FocusLost:Connect(function()
    SetSpeed(SpeedBox.Text)
end)

local DraggingSlider = false

local function UpdateSlider(inputX)

    local X = Slider.AbsolutePosition.X
    local Width = Slider.AbsoluteSize.X

    local Percent = math.clamp(
        (inputX - X) / Width,
        0,
        1
    )

    local Value = MIN_SPEED + ((MAX_SPEED - MIN_SPEED) * Percent)

    SetSpeed(Value)
end

Slider.InputBegan:Connect(function(input)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        DraggingSlider = true
        UpdateSlider(input.Position.X)
    end
end)

Knob.InputBegan:Connect(function(input)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        DraggingSlider = true
    end
end)

UserInputService.InputChanged:Connect(function(input)

    if DraggingSlider then

        if input.UserInputType == Enum.UserInputType.MouseMovement

-- SAFE CLIENT MOVEMENT
--==================================================

local function MoveToArea()

    if Moving then
        return
    end

    GetCharacter()

    if not Character or not Root or not Root.Parent then
        return
    end

    local Target = Areas[SelectedArea]

    if not Target then
        Status.Text = "Área inválida"
        Status.TextColor3 = Color3.fromRGB(255, 70, 70)
        return
    end

    Moving = true

    Dropdown.Visible = false

    Status.Text = "Indo para: " .. SelectedArea
    Status.TextColor3 = Color3.fromRGB(0, 255, 150)

    GoButton.Text = "MOVENDO..."

    local StartPivot = Character:GetPivot()

    local TargetCFrame = CFrame.new(
        Target + Vector3.new(0, 3, 0)
    )

    local Distance = (StartPivot.Position - Target).Magnitude

    local Duration = math.max(
        Distance / Speed,
        0.05
    )

    -- CFrameValue é usado em vez de Tweenar
    -- diretamente o HumanoidRootPart.
    local PivotValue = Instance.new("CFrameValue")
    PivotValue.Value = StartPivot

    CurrentTween = TweenService:Create(
        PivotValue,
        TweenInfo.new(
            Duration,
            Enum.EasingStyle.Linear,
            Enum.EasingDirection.InOut
        ),
        {
            Value = TargetCFrame
        }
    )

    MoveConnection = PivotValue.Changed:Connect(function(NewCFrame)

        if Moving and Character and Character.Parent then
            Character:PivotTo(NewCFrame)
        end
    end)

    CurrentTween.Completed:Connect(function(playbackState)

        if MoveConnection then
            MoveConnection:Disconnect()
            MoveConnection = nil
        end

        PivotValue:Destroy()
        CurrentTween = nil

        if playbackState == Enum.PlaybackState.Completed then

            if Character and Character.Parent then
                Character:PivotTo(TargetCFrame)
            end

            Status.Text = "Chegou em: " .. SelectedArea
            Status.TextColor3 = Color3.fromRGB(0, 255, 100)

        end

        Moving = false
        GoButton.Text = "GO"
    end)

    CurrentTween:Play()
end

GoButton.MouseButton1Click:Connect(function()
    MoveToArea()
end)

    if DraggingWindow then

        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then

            local Delta = input.Position - DragStart

            Main.Position = UDim2.new(
                StartPosition.X.Scale,
                StartPosition.X.Offset + Delta.X,
                StartPosition.Y.Scale,
                StartPosition.Y.Offset + Delta.Y
            )
        end
    end
end)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        DraggingWindow = false
    end
end)
