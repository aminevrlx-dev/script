local function corner(object, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius)

local function stroke(object, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.Parent = object
    return s
end

local function setSpeed(value)

    value = math.clamp(
        math.floor(tonumber(value) or tweenSpeed),
        MIN_SPEED,
        MAX_SPEED
    )

    tweenSpeed = value

    SpeedBox.Text = tostring(value)

    local percent =
        (value - MIN_SPEED) /
        (MAX_SPEED - MIN_SPEED)

    SliderFill.Size = UDim2.new(percent, 0, 1, 0)

    SliderButton.Position = UDim2.new(
        percent,
        0,
        0.5,
        0
    )
end

SpeedBox.FocusLost:Connect(function()
    setSpeed(SpeedBox.Text)
end)

local sliderDragging = false

local function updateSlider(inputX)

    local startX = SliderBack.AbsolutePosition.X
    local width = SliderBack.AbsoluteSize.X

    local percent = math.clamp(
        (inputX - startX) / width,
        0,
        1
    )

    local value =
        MIN_SPEED +
        (MAX_SPEED - MIN_SPEED) * percent

    setSpeed(value)
end

SliderButton.InputBegan:Connect(function(input)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        sliderDragging = true
    end
end)

SliderBack.InputBegan:Connect(function(input)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        updateSlider(input.Position.X)
        sliderDragging = true
    end
end)

UserInputService.InputChanged:Connect(function(input)

    if sliderDragging then

        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then

            updateSlider(input.Position.X)
        end
    end
end)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        sliderDragging = false
    end
end)

--========================================================--
-- TWEEN
--========================================================--

local function stopTween()

    if currentTween then
        currentTween:Cancel()
        currentTween = nil
    end

    tweening = false

    Go.Visible = true
    Back.Visible = true
    Stop.Visible = false

    Status.Text = "Stopped"
    Status.TextColor3 = GREY
end

local function tweenTo(target, name)

    local Character = Player.Character
    if not Character then return end

    local Root = Character:FindFirstChild("HumanoidRootPart")
    if not Root then return end

    stopTween()

    local distance = (Root.Position - target).Magnitude

    if distance <= 1 then
        Status.Text = "Already at " .. name
        return
    end

    local duration = distance / tweenSpeed

    tweening = true

    Go.Visible = false
    Back.Visible = false
    Stop.Visible = true

    Status.Text =
        "Going to " .. name ..
        " • " .. tweenSpeed .. " speed"

    Status.TextColor3 = CYAN

    currentTween = TweenService:Create(
        Root,
        TweenInfo.new(
            duration,
            Enum.EasingStyle.Linear,
            Enum.EasingDirection.Out
        ),
        {
            CFrame = CFrame.new(target)
        }
    )

    currentTween.Completed:Connect(function(state)

        if state == Enum.PlaybackState.Completed then
            Status.Text = "Arrived at " .. name
            Status.TextColor3 = GREEN
        end

        tweening = false
        currentTween = nil

        Go.Visible = true
        Back.Visible = true
        Stop.Visible = false
    end)

    currentTween:Play()
end

Go.MouseButton1Click:Connect(function()

    local target = Areas[selectedArea]

    if target then
        tweenTo(target, selectedArea)
    end
end)

Back.MouseButton1Click:Connect(function()

    tweenTo(Areas.Back, "Back")
end)

Stop.MouseButton1Click:Connect(function()
    stopTween()
end)

--========================================================--
-- DRAG
--========================================================--

local dragging = false
local dragStart
local startPos

TitleBar.InputBegan:Connect(function(input)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)

TitleBar.InputEnded:Connect(function(input)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = false
    end
end)

UserInputService.InputChanged:Connect(function(input)

    if not dragging then return end

    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then

        local delta = input.Position - dragStart

        Main.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)
