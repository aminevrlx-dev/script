local function SetSpeed(Value)
    Value = math.clamp(math.floor(tonumber(Value) or 250), 50, 1000)
    Speed = Value
    SpeedBox.Text = tostring(Value)

    local Percent = (Value - 50) / 950
    Fill.Size = UDim2.new(Percent, 0, 1, 0)
    Knob.Position = UDim2.new(Percent, -7, 0.5, -7)
end

    SetSpeed(SpeedBox.Text)
end)

local SliderDragging = false

local function UpdateSlider(X)
    local Percent = math.clamp(
        (X - Slider.AbsolutePosition.X) / Slider.AbsoluteSize.X,
        0,
        1
    )

    SetSpeed(50 + (950 * Percent))
end

local function StopTween()
    if CurrentTween then
        CurrentTween:Cancel()
        CurrentTween = nil
    end

    if MoveConnection then
        MoveConnection:Disconnect()
        MoveConnection = nil
    end

    if PivotValue then
        PivotValue:Destroy()
        PivotValue = nil
    end

    Moving = false
    GoButton.Text = "GO"
    Status.Text = "Movimento parado"
    Status.TextColor3 = Color3.fromRGB(255, 180, 0)
end

StopButton.MouseButton1Click:Connect(function()
    if Moving then
        StopTween()
    end
end)

local function MoveToArea()
    if Moving then
        return
    end

    local Character = Player.Character or Player.CharacterAdded:Wait()
    local Position = Areas[SelectedArea]

    if not Character or not Position then
        return
    end

    Moving = true
    GoButton.Text = "..."
    Status.Text = "Indo para " .. SelectedArea
    Status.TextColor3 = Color3.fromRGB(0, 220, 255)

    local StartCFrame = Character:GetPivot()
    local TargetCFrame = CFrame.new(Position)

    local Distance = (StartCFrame.Position - Position).Magnitude
    local Duration = math.max(Distance / Speed, 0.05)

    PivotValue = Instance.new("CFrameValue")
    PivotValue.Value = StartCFrame

    MoveConnection = PivotValue:GetPropertyChangedSignal("Value"):Connect(function()
        if Character and Character.Parent then
            Character:PivotTo(PivotValue.Value)
        end
    end)

    CurrentTween = TweenService:Create(
        PivotValue,
        TweenInfo.new(Duration, Enum.EasingStyle.Linear),
        {Value = TargetCFrame}
    )

    CurrentTween.Completed:Connect(function(State)
        if MoveConnection then
            MoveConnection:Disconnect()
            MoveConnection = nil
        end

        if PivotValue then
            PivotValue:Destroy()
            PivotValue = nil
        end

        CurrentTween = nil

        if State == Enum.PlaybackState.Completed then
            if Character and Character.Parent then
                Character:PivotTo(TargetCFrame)
            end

            Status.Text = "Chegou em " .. SelectedArea
            Status.TextColor3 = Color3.fromRGB(0, 255, 100)

            Notify(
                "Chegou!",
                "Você chegou em " .. SelectedArea .. ".",
                "success"
            )
        end

        Moving = false
        GoButton.Text = "GO"
    end)

    CurrentTween:Play()
end

GoButton.MouseButton1Click:Connect(function()
    MoveToArea()
end)

BackButton.MouseButton1Click:Connect(function()
    SelectedArea = "Back"
    AreaButton.Text = "Back  ▼"

ProximityPromptService.PromptTriggered:Connect(function(Prompt, TriggeringPlayer)
    if TriggeringPlayer and TriggeringPlayer ~= Player then
        return
    end

    LastPrompt = Prompt
    LastPromptPosition = Prompt:GetPivot()
    AutoPrompt = true

    Notify(
        "Ovo detectado!",
        "Último ProximityPrompt salvo.",
        "success"
    )
end)

local function FindLastPrompt()
    if LastPrompt and LastPrompt.Parent then
        return LastPrompt
    end

    return nil
end

local function ReturnToLastPrompt()
    if Reconnecting or not AutoPrompt then
        return
    end

    local Prompt = FindLastPrompt()

    if not Prompt or not LastPromptPosition then
        return
    end

    Reconnecting = true

    task.wait(1)

    local Character = Player.Character or Player.CharacterAdded:Wait()

    if Character and Character.Parent then
        Character:PivotTo(LastPromptPosition)
    end

    task.wait(0.5)

    if Prompt and Prompt.Parent then
        pcall(function()
            fireproximityprompt(Prompt)
        end)
    end

    Reconnecting = false
end

Player.CharacterAdded:Connect(function()
    if AutoPrompt and LastPrompt then
        task.spawn(function()
            ReturnToLastPrompt()
        end)
    end
end)

                StartPosition.X.Scale,
                StartPosition.X.Offset + Delta.X,
                StartPosition.Y.Scale,
                StartPosition.Y.Offset + Delta.Y

end)

    if Input.UserInputType == Enum.UserInputType.MouseButton1
        or Input.UserInputType == Enum.UserInputType.Touch then

        DraggingWindow = false
    end
end)
