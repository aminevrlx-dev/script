--========================================================--
-- TWEEN FUNCTION
--========================================================--

local function stopTween()

    if currentTween then
        currentTween:Cancel()
        currentTween = nil
    end

    tweening = false

    GoButton.Visible = true
    BackButton.Visible = true
    StopButton.Visible = false

    Status.Text = "Stopped"
    Status.TextColor3 = GREY
end

local function tweenTo(position, areaName)

    local Character = Player.Character

    if not Character then
        Status.Text = "Character not found"
        return
    end

    local Root = Character:FindFirstChild("HumanoidRootPart")

    if not Root then
        Status.Text = "HumanoidRootPart not found"
        return
    end

    stopTween()

    local distance = (Root.Position - position).Magnitude

    if distance < 1 then
        Status.Text = "Already at " .. areaName
        return
    end

    local duration = distance / SPEED

    tweening = true

    GoButton.Visible = false
    BackButton.Visible = false
    StopButton.Visible = true

    Status.Text = "Tweening to " .. areaName .. "..."
    Status.TextColor3 = CYAN

    currentTween = TweenService:Create(
        Root,
        TweenInfo.new(
            duration,
            Enum.EasingStyle.Linear,
            Enum.EasingDirection.Out
        ),
        {
            CFrame = CFrame.new(position)
        }
    )

    currentTween.Completed:Connect(function(playbackState)

        if playbackState == Enum.PlaybackState.Completed then
            Status.Text = "Arrived at " .. areaName
            Status.TextColor3 = GREEN
        end

        tweening = false
        currentTween = nil

        GoButton.Visible = true
        BackButton.Visible = true
        StopButton.Visible = false
    end)

    currentTween:Play()
end

end)

StopButton.MouseButton1Click:Connect(function()

end)

--========================================================--
-- DRAG SYSTEM
--========================================================--

local dragging = false
local dragStart
local startPosition

TitleBar.InputBegan:Connect(function(input)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPosition = Main.Position

        input.Changed:Connect(function()

            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)

    if not dragging then
        return
    end

    if input.UserInputType ~= Enum.UserInputType.MouseMovement
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    local delta = input.Position - dragStart

    Main.Position = UDim2.new(
        startPosition.X.Scale,
        startPosition.X.Offset + delta.X,
        startPosition.Y.Scale,
        startPosition.Y.Offset + delta.Y
    )
end)
